# Separate receive-only diagnostic. Generated bootloader and table are build inputs only.
cmake_minimum_required(VERSION 3.16)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
option(K151_QUESTION_TRIAL "Enable one question capture after wake in the receive-only trial" OFF)
option(K151_WAKE_ONLY_CONTROL "Enable the short post-wake observation without a question" OFF)
if(K151_QUESTION_TRIAL AND K151_WAKE_ONLY_CONTROL)
    message(FATAL_ERROR "Select either the question trial or the wake-only control")
endif()
if(K151_WAKE_ONLY_CONTROL)
    project(k151_wake_only_control)
elseif(K151_QUESTION_TRIAL)
    project(k151_mic_question_trial)
else()
    project(k151_mic_wake_trial)
endif()
idf_build_get_property(trial_components BUILD_COMPONENTS)
foreach(forbidden nvs_flash esp_wifi esp_netif esp_http_client esp_tls)
    if(forbidden IN_LIST trial_components)
        message(FATAL_ERROR "Mic wake trial must not link ${forbidden}")
    endif()
endforeach()
